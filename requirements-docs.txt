# What Vercel installs to build the documentation site.
#
# This duplicates the `docs` dependency group in pyproject.toml, which is what `uv sync
# --group docs` reads locally. The duplication exists because Vercel's build image runs
# pip against a requirements file and does not read PEP 735 dependency groups.
# tests/test_docs_requirements.py fails if the two lists drift apart.
#
# Pinned rather than floored: a docs build that silently picks up a new major version of
# the theme is a site whose layout changed without a commit saying so.
mkdocs==1.6.1
mkdocs-material==9.7.7
mkdocstrings[python]==1.0.6
mkdocstrings-python==2.0.7

# Not a documentation tool here. mkdocstrings shells out to it to format the signatures
# on the Reference page, and without it Vercel would render signatures unformatted while
# a local build (where ruff is in the dev group) renders them formatted. The production
# site would then differ from the one the author reviewed.
ruff==0.16.4
